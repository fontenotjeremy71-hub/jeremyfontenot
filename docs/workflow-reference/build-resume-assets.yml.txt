name: Build Resume Assets

on:
  workflow_dispatch:
  push:
    branches: [main]
    paths:
      - scripts/build/generate-resume-assets.py

permissions:
  contents: write

jobs:
  build-resume:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.12'

      - name: Install resume tooling
        run: |
          python -m pip install --disable-pip-version-check python-docx
          sudo apt-get update
          sudo apt-get install -y --no-install-recommends libreoffice-writer

      - name: Generate DOCX
        run: python scripts/build/generate-resume-assets.py

      - name: Generate PDF
        run: |
          mkdir -p .resume-pdf
          libreoffice --headless --convert-to pdf --outdir .resume-pdf assets/resume/jeremy-fontenot-resume.docx
          mv .resume-pdf/jeremy-fontenot-resume.pdf assets/resume/jeremy-fontenot-resume.pdf

      - name: Commit updated resume assets
        shell: bash
        run: |
          if git diff --quiet -- assets/resume/jeremy-fontenot-resume.docx assets/resume/jeremy-fontenot-resume.pdf; then
            echo "Resume assets are already current."
            exit 0
          fi
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add assets/resume/jeremy-fontenot-resume.docx assets/resume/jeremy-fontenot-resume.pdf
          git commit -m "Update downloadable resume assets"
          git push origin HEAD:main
